    [STAThread]
    private static int Main(string[] args)
    {
        try
        {
            if (args.Contains("--remaining", StringComparer.OrdinalIgnoreCase)) return ExportRemaining(args);
            Root = Path.GetFullPath(args.Length > 0 ? args[0] : Path.Combine(AppContext.BaseDirectory, "..", "..", "..", "..", ".."));
            if (!File.Exists(Path.Combine(Root, "alpha-island.csproj"))) throw new InvalidDataException("The project directory is required.");
            Out = Path.Combine(Root, "bin", "IconReview-2026-10-11-v" + typeof(SemanticIcon).Assembly.GetName().Version!.ToString(3));
            Directory.CreateDirectory(Out);
            _ = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
            MotionPolicy.OverrideRenderTierForDebug(2); MotionPolicy.Apply(false, false);
            ExportOriginalAssets();
            ExportSemanticIcons();
            ExportFontGlyphs();
            ExportSources();
            WriteIndex();
            WriteOverview();
            WriteMotionSequence();
            var counts = Entries.GroupBy(e => e.Category).Select(g => new { Category = g.Key, Icons = g.Count(), Animated = g.Count(e => e.AnimationPath != null), Changing = g.Count(e => e.HasVisualMotion == true) }).ToArray();
            File.WriteAllText(Path.Combine(Out, "manifest.json"), JsonSerializer.Serialize(new
            {
                FormatVersion = 1, Version = typeof(SemanticIcon).Assembly.GetName().Version?.ToString(),
                ExportedAt = DateTimeOffset.Now.ToString("O"), Render = "Actual production WPF SemanticIcon and ActionIconMotion, captured at 40 ms intervals",
                Entries, Counts = counts, Files = Directory.EnumerateFiles(Out, "*", SearchOption.AllDirectories)
                    .Count(file => !string.Equals(file, Path.Combine(Out, "manifest.json"), StringComparison.OrdinalIgnoreCase)) + 1
            }, new JsonSerializerOptions { WriteIndented = true }));
            foreach (var count in counts) Console.WriteLine($"{count.Category}: {count.Icons} icons, {count.Animated} GIFs");
            Console.WriteLine($"TOTAL {Entries.Count} icons, {Entries.Count(e => e.AnimationPath != null)} GIFs, {Directory.EnumerateFiles(Out, "*", SearchOption.AllDirectories).Count()} files");
            Console.WriteLine(Out);
            return 0;
        }
        catch (Exception e) { Console.Error.WriteLine($"Icon export failed: {e.GetType().Name}: {e.Message}"); return 1; }
    }

    private static void ExportSemanticIcons()
    {
        var specs = Specs();
        int columns = 8, rows = (int)Math.Ceiling(specs.Count / (double)columns);
        int width = columns * Tile, height = rows * Tile;
        var canvas = new Canvas { Width = width, Height = height, Background = Brushes.Transparent };
        var icons = new List<SemanticIcon>();
        var rectangles = new List<Int32Rect>();
        foreach (var spec in specs)
        {
            int n = icons.Count, x = n % columns * Tile, y = n / columns * Tile;
            var icon = new SemanticIcon { Width = Tile, Height = Tile, Shape = spec.Shape, Text = spec.Text,
                Active = spec.Active, Level = spec.Level, WeatherCode = spec.Weather,
                Foreground = Cyan, FontFamily = new FontFamily("Segoe MDL2 Assets") };
            if (spec.Device != null)
            {
                var artwork = typeof(SemanticIcon).Assembly.GetType("alpha_island.Controls.BluetoothDeviceArtwork")!;
                icon.Source = (ImageSource?)artwork.GetMethod("For", BindingFlags.Static | BindingFlags.NonPublic)!.Invoke(null, new object?[] { spec.Device });
            }
            Canvas.SetLeft(icon, x); Canvas.SetTop(icon, y); canvas.Children.Add(icon);
            icons.Add(icon); rectangles.Add(new(x, y, Tile, Tile));
        }
        using var source = new HwndSource(new HwndSourceParameters("Alpha Island icon export")
        { Width = width, Height = height, PositionX = -32000, PositionY = -32000, WindowStyle = unchecked((int)0x80000000) });
        source.RootVisual = canvas; canvas.UpdateLayout(); Pump(40);
        var frames = new List<BitmapSource>[icons.Count];
        var staticFrame = Render(canvas, width, height);
        for (int n = 0; n < icons.Count; n++)
        {
            var spec = specs[n]; string path = $"{spec.Category}/{spec.Id}";
            SavePng(new CroppedBitmap(staticFrame, rectangles[n]), path + "/静态.png");
            frames[n] = new() { new CroppedBitmap(staticFrame, rectangles[n]) };
        }
        // Capture the real finite ActionIconMotion clocks. No hand-coded drawing or interpolation.
        foreach (bool target in new[] { false, true })
        {
            for (int n = 0; n < icons.Count; n++)
            {
                icons[n].Active = target;
                icons[n].Level = target ? specs[n].Level : specs[n].Shape == IconShape.Volume ? 0 : .2;
                ActionIconMotion.Play(icons[n], target ? ActionIconAnimation.Connect : ActionIconAnimation.Disconnect);
            }
            for (int step = 0; step < 40; step++)
            {
                Pump(40); canvas.UpdateLayout(); var shot = Render(canvas, width, height);
                for (int n = 0; n < icons.Count; n++) frames[n].Add(new CroppedBitmap(shot, rectangles[n]));
            }
        }
        for (int n = 0; n < icons.Count; n++)
        {
            var spec = specs[n]; string path = $"{spec.Category}/{spec.Id}";
            // Keep transparent full-resolution frames for redesign; GIF gets theme background.
            for (int frame = 0; frame < frames[n].Count; frame++) SavePng(frames[n][frame], path + $"/PNG帧/frame-{frame:D3}.png");
            SaveGif(frames[n].Select(ThemeFrame).ToArray(), path + "/状态过渡.gif");
            string note = "实际程序语义控件与动画；GIF 含关闭和开启动作。仅预览循环，程序里动画仅播放一次。";
            bool changes = frames[n].Select(FrameDigest).Distinct().Take(2).Count() > 1;
            if (!changes) note += " 当前生产状态的内部图形保持不变，GIF 如实保留静态状态，没有人为增加动画。";
            if (spec.Shape == IconShape.Volume) note += " 音量变化 180 ms；静音切换 320 ms；档位声波逐条变化。";
            if (spec.Shape == IconShape.Agent && spec.Text != "\uE99A") note += " 该状态图形按当前 Text 保留。";
            Entries.Add(new(spec.Category, spec.Id, spec.Title, note, path + "/静态.png", path + "/状态过渡.gif", path + "/PNG帧", new[] { "Controls/SemanticIcon.cs", spec.Shape == IconShape.Weather ? "Controls/SemanticIcon.Weather.cs" : "Animation/ActionIconMotion.cs" }, changes));
            ActionIconMotion.Stop(icons[n]);
        }
        source.RootVisual = null;
    }

    private static void ExportFontGlyphs()
    {
        string[] dirs = { "Animation", "Controls", "Core", "Interop", "Services", "Settings", "Themes", "Plugins", "ConversationCore", "examples" };
        var files = new List<string>();
        files.AddRange(Directory.GetFiles(Root, "MainWindow*.cs"));
        files.AddRange(new[] { "MainWindow.xaml", "App.xaml" }.Select(f => Path.Combine(Root, f)));
        foreach (string dir in dirs)
            if (Directory.Exists(Path.Combine(Root, dir))) files.AddRange(Directory.EnumerateFiles(Path.Combine(Root, dir), "*", SearchOption.AllDirectories)
                .Where(f => (f.EndsWith(".cs") || f.EndsWith(".xaml")) && !f.Contains("\\bin\\") && !f.Contains("\\obj\\") && !f.Contains("\\upstream\\")));
        var glyphs = new SortedDictionary<int, HashSet<string>>();
        var pattern = new Regex(@"(?:\\u|&#x)([EeFf][0-9a-fA-F]{3})", RegexOptions.CultureInvariant);
        foreach (string file in files.Distinct())
        {
            string relative = Path.GetRelativePath(Root, file).Replace('\\', '/');
            string[] lines = File.ReadAllLines(file);
            for (int line = 0; line < lines.Length; line++)
                foreach (Match match in pattern.Matches(lines[line]))
                {
                    int code = int.Parse(match.Groups[1].Value, NumberStyles.HexNumber, CultureInfo.InvariantCulture);
                    if (code > 0xF8FF) continue; // Full-width punctuation and replacement markers are not icon glyphs.
                    if (!glyphs.TryGetValue(code, out var locations)) glyphs.Add(code, locations = new());
                    locations.Add(relative + ":" + (line + 1));
                }
        }
        var typeface = new Typeface(new FontFamily("Segoe MDL2 Assets"), FontStyles.Normal, FontWeights.Normal, FontStretches.Normal);
        if (!typeface.TryGetGlyphTypeface(out var glyphTypeface)) throw new InvalidDataException("Segoe MDL2 Assets cannot be resolved.");
        var unavailable = new List<string>();
        foreach (var (code, sources) in glyphs)
        {
            string glyph = char.ConvertFromUtf32(code);
            bool vector = AlphaIsland.PluginSdk.GlyphArtwork.Titles.TryGetValue(glyph, out var title);
            if (!vector && !glyphTypeface.CharacterToGlyphMap.ContainsKey(code))
            {
                unavailable.Add("U-" + code.ToString("X4") + "：本机字体没有该码点；可能是校验范围、旧码点或由语义控件替代。来源：" + string.Join("；", sources));
                foreach (string staleCategory in new[] { "07-设置字体图标", "08-插件与对话字体图标", "09-通用控件字体图标" })
                {
                    string stale = Path.Combine(Out, staleCategory, "U-" + code.ToString("X4") + ".png");
                    if (File.Exists(stale)) File.Delete(stale); // Only this exporter's named PNG, never directories.
                }
                continue;
            }
            string category = GlyphCategory(sources);
            string id = "U-" + code.ToString("X4");
            var text = new AlphaIsland.PluginSdk.GlyphIcon { Text = glyph, Width = Tile, Height = Tile,
                FontFamily = new FontFamily("Segoe MDL2 Assets"), FontSize = 66, Foreground = Cyan,
                TextAlignment = TextAlignment.Center, VerticalAlignment = VerticalAlignment.Center };
            var panel = new Grid { Width = Tile, Height = Tile, Background = Brushes.Transparent };
            panel.Children.Add(text); panel.Measure(new(Tile, Tile)); panel.Arrange(new(0, 0, Tile, Tile)); panel.UpdateLayout();
            SavePng(Render(panel, Tile, Tile), category + "/" + id + ".png");
            if (vector) File.WriteAllText(Path.Combine(Out, category, id + ".svg"), AlphaIsland.PluginSdk.GlyphArtwork.ExportSvg(glyph), new UTF8Encoding(false));
            Entries.Add(new(category, id, title ?? id, vector ? "原创圆角矢量图标；保留原 glyph ID 兼容插件，主程序和插件使用同一绘制目录。" : "第三方/备用字体字形。", category + "/" + id + ".png", Sources: sources.Order().ToArray()));
        }
        File.WriteAllLines(Path.Combine(Out, "非字体字形说明.txt"), unavailable, new UTF8Encoding(false));
    }

    private static string GlyphCategory(IEnumerable<string> sources)
    {
        string combined = string.Join(" ", sources);
        if (combined.Contains("Settings/") || combined.Contains("Win11Settings")) return "07-设置图标";
        if (combined.Contains("examples/") || combined.Contains("Plugins/") || combined.Contains("ConversationCore/")) return "08-插件与对话图标";
        return "09-通用控件图标";
    }

    private static BitmapSource ThemeFrame(BitmapSource image)
    {
        var visual = new DrawingVisual();
        using (var dc = visual.RenderOpen())
        { dc.DrawRectangle(new SolidColorBrush(Color.FromRgb(0x18, 0x22, 0x37)), null, new Rect(0, 0, Tile, Tile)); dc.DrawImage(image, new Rect(0, 0, Tile, Tile)); }
        var bitmap = new RenderTargetBitmap(Tile, Tile, 96, 96, PixelFormats.Pbgra32); bitmap.Render(visual); bitmap.Freeze(); return bitmap;
    }
    private static string FrameDigest(BitmapSource image)
    {
        int stride = (image.PixelWidth * image.Format.BitsPerPixel + 7) / 8;
        var pixels = new byte[stride * image.PixelHeight]; image.CopyPixels(pixels, stride, 0);
        return Convert.ToHexString(SHA256.HashData(pixels));
    }
    private static void SaveGif(IReadOnlyList<BitmapSource> frames, string relative)
    {
        var encoder = new GifBitmapEncoder();
        foreach (var frame in frames)
        {
            var metadata = new BitmapMetadata("gif"); metadata.SetQuery("/grctlext/Delay", (ushort)4); metadata.SetQuery("/grctlext/Disposal", (byte)1);
            encoder.Frames.Add(BitmapFrame.Create(frame, null, metadata, null));
        }
        using var memory = new MemoryStream(); encoder.Save(memory); byte[] bytes = memory.ToArray();
        int palette = (bytes[10] & 0x80) != 0 ? 3 * (1 << ((bytes[10] & 7) + 1)) : 0, at = 13 + palette;
        byte[] loop = { 0x21, 0xFF, 0x0B, 78, 69, 84, 83, 67, 65, 80, 69, 50, 46, 48, 3, 1, 0, 0, 0 };
        using (var file = File.Create(Path.Combine(Out, relative))) { file.Write(bytes, 0, at); file.Write(loop); file.Write(bytes, at, bytes.Length - at); }
        using var check = File.OpenRead(Path.Combine(Out, relative));
        var decoded = new GifBitmapDecoder(check, BitmapCreateOptions.PreservePixelFormat, BitmapCacheOption.OnLoad);
        if (decoded.Frames.Count != frames.Count || decoded.Frames.Any(f => f.PixelWidth != Tile || f.PixelHeight != Tile)) throw new InvalidDataException("GIF verification failed.");
    }
    private static RenderTargetBitmap Render(Visual visual, int width, int height)
    {
        var image = new RenderTargetBitmap(width, height, 96, 96, PixelFormats.Pbgra32); image.Render(visual); image.Freeze(); return image;
    }
    private static void Pump(int milliseconds)
    {
        var frame = new DispatcherFrame();
        var timer = new DispatcherTimer(DispatcherPriority.Background) { Interval = TimeSpan.FromMilliseconds(milliseconds) };
        timer.Tick += (_, _) => { timer.Stop(); frame.Continue = false; };
        timer.Start(); Dispatcher.PushFrame(frame);
    }
