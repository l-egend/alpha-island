    {
        foreach (var spec in specs)

        var canvas = new Canvas { Width = width, Height = height, Background = Brushes.Transparent };

        for (int index = 0; index < icons.Length; index++)
        {
            Canvas.SetLeft(icons[index], index % columns * Tile); Canvas.SetTop(icons[index], index / columns * Tile);
            canvas.Children.Add(icons[index]);
        }

        var frames = icons.Select(_ => new List<BitmapSource>()).ToArray();

        {
            canvas.UpdateLayout(); var shot = Render(canvas, width, height);
            for (int index = 0; index < icons.Length; index++)
            {
                var frame = new CroppedBitmap(shot, new Int32Rect(index % columns * Tile, index / columns * Tile, Tile, Tile));

    private static void ExportRemainingGlyphs()
    {
        var locations = new SortedDictionary<int, SortedSet<string>>();
        string[] roots = { "Animation", "Controls", "Core", "Interop", "Services", "Settings", "Themes", "Plugins", "ConversationCore", "AgentTools", "examples" };
        var files = Directory.EnumerateFiles(Root, "MainWindow*", SearchOption.TopDirectoryOnly)
            .Concat(new[] { "App.xaml" }.Select(file => Path.Combine(Root, file)))
            .Concat(roots.SelectMany(dir => Directory.EnumerateFiles(Path.Combine(Root, dir), "*", SearchOption.AllDirectories)))
            .Where(file => file.EndsWith(".cs", StringComparison.OrdinalIgnoreCase) || file.EndsWith(".xaml", StringComparison.OrdinalIgnoreCase))
            .Where(file => !Path.GetRelativePath(Root, file).Split(Path.DirectorySeparatorChar).Any(part => part is "bin" or "obj" or "upstream" or "node_modules")).Distinct();
        var pattern = new Regex(@"(?:\\u|&#x)([EeFf][0-9a-fA-F]{3})|([\uE000-\uF8FF])", RegexOptions.CultureInvariant);
        foreach (string file in files)
        {
            string[] lines = File.ReadAllLines(file);
            for (int line = 0; line < lines.Length; line++)
                foreach (Match match in pattern.Matches(lines[line]))
                {
                    int code = match.Groups[1].Success ? int.Parse(match.Groups[1].Value, NumberStyles.HexNumber) : match.Groups[2].Value[0];
                    if (!locations.TryGetValue(code, out var sources)) locations[code] = sources = new(StringComparer.Ordinal);
                    sources.Add(Path.GetRelativePath(Root, file).Replace('\\', '/') + ":" + (line + 1));
                }
        }
        var typeface = new Typeface(new FontFamily("Segoe MDL2 Assets"), FontStyles.Normal, FontWeights.Normal, FontStretches.Normal);
        if (!typeface.TryGetGlyphTypeface(out var glyphTypeface)) throw new InvalidDataException("Segoe MDL2 Assets is unavailable.");
        foreach (var (code, sources) in locations)
        {
            string glyph = char.ConvertFromUtf32(code), id = $"U-{code:X4}";
            if (GlyphArtwork.Titles.TryGetValue(glyph, out string? title))
            {
                Omitted.Add(new { Kind = "Glyph", Id = id, Title = title, Reason = "0.1.40 已统一重绘；后续已改过的删除、网速、屏幕翻译、电池等仍排除", Sources = sources });
                continue;
            }
            if (code is 0xE000 or 0xF8FF || code > 0xF8FF)
            { NotIcons.Add(id + "：范围校验标记或非图标字符。来源：" + string.Join("; ", sources)); continue; }
            if (code == 0xE9BD)
            { NotIcons.Add(id + "：天气入口实际使用 SemanticIcon 的天气路径，已逐种天气导出。来源：" + string.Join("; ", sources)); continue; }
            if (!glyphTypeface.CharacterToGlyphMap.ContainsKey(code))
            { NotIcons.Add(id + "：本机 Segoe MDL2 Assets 未定义此字形，不将缺字方框作为素材。来源：" + string.Join("; ", sources)); continue; }
            string category = GlyphCategory(sources), stem = category + "/" + id;
            var icon = new GlyphIcon { Text = glyph, Width = ReviewSize, Height = ReviewSize,
                FontFamily = new FontFamily("Segoe MDL2 Assets"), Foreground = Cyan };
            Layout(icon, ReviewSize, ReviewSize);
            SavePng(Render(icon, ReviewSize, ReviewSize), stem + ".png"); SaveActualSvg(icon, stem + ".svg");
            title = code switch { 0xE898 => "上传 / 上行流量提示", 0xE917 => "时钟插件", _ => id };
            Entries.Add(new(category, id, title, "实际字体图标已转为可编辑轮廓，无需安装字体即可查看 SVG。", stem + ".png", Sources: sources.ToArray()));
        }
    }

    private static void SaveActualSvg(Visual visual, string relative)
    {
        var drawing = VisualTreeHelper.GetDrawing(visual) ?? throw new InvalidDataException("Icon has no WPF drawing.");
        XNamespace ns = "http://www.w3.org/2000/svg";
        var root = new XElement(ns + "svg", new XAttribute("viewBox", $"0 0 {ReviewSize} {ReviewSize}"));
        var definitions = new XElement(ns + "defs"); root.Add(definitions);
        string Number(double value) => value.ToString("0.#########", CultureInfo.InvariantCulture);
        string Matrix(Matrix value) => $"matrix({Number(value.M11)} {Number(value.M12)} {Number(value.M21)} {Number(value.M22)} {Number(value.OffsetX)} {Number(value.OffsetY)})";
        XElement Shape(Geometry geometry, SolidColorBrush brush)
        {
            var path = PathGeometry.CreateFromGeometry(geometry);
            string data = path.ToString(CultureInfo.InvariantCulture);
            if (data.StartsWith("F0", StringComparison.Ordinal) || data.StartsWith("F1", StringComparison.Ordinal)) data = data[2..].TrimStart();
            var item = new XElement(ns + "path", new XAttribute("d", data), new XAttribute("fill", $"#{brush.Color.R:X2}{brush.Color.G:X2}{brush.Color.B:X2}"),
                new XAttribute("fill-rule", path.FillRule == FillRule.EvenOdd ? "evenodd" : "nonzero"));
            double opacity = brush.Opacity * brush.Color.A / 255d;
            if (opacity != 1) item.Add(new XAttribute("opacity", Number(opacity)));
            if (path.Transform != null && !path.Transform.Value.IsIdentity) item.Add(new XAttribute("transform", Matrix(path.Transform.Value)));
            return item;
        }
        void AddDrawing(Drawing node, XElement parent)
        {
            if (node is DrawingGroup group)
            {
                var container = new XElement(ns + "g"); parent.Add(container);
                if (group.Transform != null && !group.Transform.Value.IsIdentity) container.Add(new XAttribute("transform", Matrix(group.Transform.Value)));
                if (group.Opacity != 1) container.Add(new XAttribute("opacity", Number(group.Opacity)));
                if (group.ClipGeometry != null)
                {
                    string id = "clip" + definitions.Elements().Count(); var clip = new XElement(ns + "clipPath", new XAttribute("id", id));
                    clip.Add(Shape(group.ClipGeometry, Brushes.White)); definitions.Add(clip); container.Add(new XAttribute("clip-path", $"url(#{id})"));
                }
                foreach (Drawing child in group.Children) AddDrawing(child, container);
            }
            else if (node is GeometryDrawing shape)
            {
                if (shape.Brush is SolidColorBrush fill) parent.Add(Shape(shape.Geometry, fill));
                if (shape.Pen?.Brush is SolidColorBrush ink) parent.Add(Shape(shape.Geometry.GetWidenedPathGeometry(shape.Pen), ink));
            }
            else if (node is GlyphRunDrawing text && text.ForegroundBrush is SolidColorBrush ink) parent.Add(Shape(text.GlyphRun.BuildGeometry(), ink));
            else throw new InvalidDataException("Unsupported SVG drawing: " + node.GetType().Name);
        }
        AddDrawing(drawing, root);
        File.WriteAllText(Path.Combine(Out, relative), new XDocument(root).ToString(), new UTF8Encoding(false));
    }

    private static void VerifyRemainingReview()
    {
        foreach (var entry in Entries)
        {
            var image = Load(entry.StaticPath);
            int stride = image.PixelWidth * 4; var pixels = new byte[stride * image.PixelHeight];
            new FormatConvertedBitmap(image, PixelFormats.Pbgra32, null, 0).CopyPixels(pixels, stride, 0);
            if (!Enumerable.Range(0, image.PixelWidth * image.PixelHeight).Any(index => pixels[index * 4 + 3] > 40))
                throw new InvalidDataException("Empty PNG: " + entry.Id);
            string svg = Path.Combine(Out, SvgFor(entry));
            if (File.Exists(svg)) foreach (var path in XDocument.Load(svg).Descendants().Where(item => item.Name.LocalName == "path")) _ = Geometry.Parse((string)path.Attribute("d")!);
            if (entry.AnimationPath != null && !File.Exists(Path.Combine(Out, entry.AnimationPath))) throw new FileNotFoundException(entry.AnimationPath);
            if (PreviouslyReviewedSemantic.Contains(entry.Id) || entry.Id.StartsWith("U-", StringComparison.Ordinal) && GlyphArtwork.Titles.ContainsKey(char.ConvertFromUtf32(Convert.ToInt32(entry.Id[2..], 16))))
                throw new InvalidDataException("A previously reviewed icon leaked into the remaining set.");
        }
    }
